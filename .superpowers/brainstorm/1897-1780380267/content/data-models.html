resumes collection — Resume document

highlights: string[] } ]

startDate, endDate, score } ]
skills: [ { name, level, keywords: string[] } ]
certificates / awards / languages: [ ... ]
volunteer / publications / projects: [ ... ]
}
meta: { design state — not in JSON Resume spec
templateId: string
fontFamily / headerFontFamily: string
primaryColor / accentColor: string #hex
pageMargins: number 0.5–1.5 inches
lineSpacing: number 1.0–1.15
sectionOrder: string[] drag-and-drop order
layout: 'single-column' | 'two-column'
}
users collection — NextAuth managed
_id: ObjectId
name / email / image: string
emailVerified: Date | null
accounts, sessions — managed by @auth/mongoose-adapter
